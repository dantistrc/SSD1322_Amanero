// ============================================================
//          СОСТОЯНИЯ МЕНЮ АППАРАТА (ДОБАВИТЬ В MAIN.C)
// ============================================================

typedef enum {
    MENU_STATE_SPLASH = 0, // Заставка при включении (1 сек)
    MENU_STATE_VOLUME,     // Главный экран (Громкость, входы, поток)
    MENU_STATE_INPUT,      // Экран выбора входа (USB, SPDIF, TOSLINK)
    MENU_STATE_FILTER,     // Экран выбора фильтра ЦАПа
    MENU_STATE_BALANCE     // Экран настройки баланса каналов
} MenuState_t;

// Глобальные переменные управления
MenuState_t current_menu_state = MENU_STATE_SPLASH;

uint32_t menu_timer = 0;     // Таймер для автовозврата в VOLUME (60 сек)
uint32_t dimming_timer = 0;  // Таймер для падения яркости экрана
uint8_t  screen_dimmed = 0;  // Флаг: экран сейчас затенён или горит ярко

// Настройки аппарата (в будущем будут грузиться из памяти)
uint8_t  current_volume = 40;  // Значение громкости в дБ
uint8_t  current_input = 0;   // 0 - USB, 1 - SPDIF, 2 - TOSLINK
uint8_t  current_filter = 0;  // Индекс активного фильтра
int8_t   current_balance = 0; // Баланс: 0 - центр, минус - лево, плюс - право

// ============================================================
//            ГЛАВНЫЙ ЦИКЛ ОБРАБОТКИ МЕНЮ В WHILE(1)
// ============================================================
void Menu_Process(void) {
    
    // Если экран затенён, а пользователь что-то нажал или крутанул
    if (screen_dimmed && (Encoder_Moved() || Button_Pressed() || IR_Action())) {
        SSD1322_SetBrightness(100); // Возвращаем 100% яркость
        screen_dimmed = 0;
        dimming_timer = GetTick();  // Сбрасываем таймер затенения
        menu_timer = GetTick();     // Сбрасываем таймер автовозврата
        return;                     // Первое действие только пробуждает экран
    }

    switch (current_menu_state) {
        
        // 1. ЗАСТАВКА ПРИ СТАРТЕ
        case MENU_STATE_SPLASH:
            Draw_SplashScreen(); // Показываем логотип, чип, серийник
            delay_ms(1000);      // Держим 1 секунду
            
            // Загружаем сохраненные настройки из памяти (сделаем позже)
            // Load_Settings_From_Flash(); 
            
            current_menu_state = MENU_STATE_VOLUME; // Переходим на главный экран
            SSD1322_Clear();
            break;

        // 2. ГЛАВНЫЙ ЭКРАН (VOLUME)
        case MENU_STATE_VOLUME:
            Draw_VolumeScreen(); // Отрисовка: большие цифры, вверху входы, поток
            
            // Обработка энкодера в режиме громкости
            if (Encoder_Rotated_Left())  { current_volume -= 2; }
            if (Encoder_Rotated_Right()) { current_volume += 2; }
            
            // Короткий клик переводит в меню входов
            if (Encoder_Button_Pressed()) {
                current_menu_state = MENU_STATE_INPUT;
                menu_timer = GetTick(); // Запускаем счетчик 60 секунд
                SSD1322_Clear();
            }
            break;

        // 3. ЭКРАН ВЫБОРА ВХОДА (INPUT)
        case MENU_STATE_INPUT:
            Draw_InputScreen(); // Крупно по центру выбранный вход
            
            // Крутим с упорами (0 - USB, 1 - SPDIF, 2 - TOSLINK)
            if (Encoder_Rotated_Left() && current_input > 0)  { current_input--;  menu_timer = GetTick(); }
            if (Encoder_Rotated_Right() && current_input < 2) { current_input++; menu_timer = GetTick(); }
            
            // Проверка автовозврата по таймауту (60 секунд)
            if (GetTick() - menu_timer > 60000) {
                current_menu_state = MENU_STATE_VOLUME;
                SSD1322_Clear();
            }
            
            // Клик переводит к фильтрам
            if (Encoder_Button_Pressed()) {
                current_menu_state = MENU_STATE_FILTER;
                menu_timer = GetTick();
                SSD1322_Clear();
            }
            break;

        // 4. ЭКРАН ВЫБОРА ФИЛЬТРА (FILTER)
        case MENU_STATE_FILTER:
            Draw_FilterScreen(); // Крупно по центру название фильтра
            
            // Переключение фильтров энкодером
            if (Encoder_Rotated_Left())  { current_filter--; menu_timer = GetTick(); }
            if (Encoder_Rotated_Right()) { current_filter++; menu_timer = GetTick(); }
            
            if (GetTick() - menu_timer > 60000) {
                current_menu_state = MENU_STATE_VOLUME;
                SSD1322_Clear();
            }
            
            // Клик переводит к балансу
            if (Encoder_Button_Pressed()) {
                current_menu_state = MENU_STATE_BALANCE;
                menu_timer = GetTick();
                SSD1322_Clear();
            }
            break;

        // 5. ЭКРАН НАСТРОЙКИ БАЛАНСА (BALANCE)
        case MENU_STATE_BALANCE:
            Draw_BalanceScreen(); // Крупно шкала баланса и децибелы
            
            if (Encoder_Rotated_Left())  { current_balance--; menu_timer = GetTick(); }
            if (Encoder_Rotated_Right()) { current_balance++; menu_timer = GetTick(); }
            
            if (GetTick() - menu_timer > 60000) {
                current_menu_state = MENU_STATE_VOLUME;
                SSD1322_Clear();
            }
            
            // Клик возвращает на главный экран
            if (Encoder_Button_Pressed()) {
                current_menu_state = MENU_STATE_VOLUME;
                SSD1322_Clear();
            }
            break;
    }
    
    // Логика автоматического затенения экрана (например, через 3 минуты простоя)
    if (!screen_dimmed && (GetTick() - dimming_timer > 180000)) {
        SSD1322_SetBrightness(15); // Гасим яркость до 15%
        screen_dimmed = 1;
    }
}

void SSD1322_DrawCustomChar(uint16_t x, uint16_t y, char c) {
    // Проверяем, что символ входит в рамки таблицы (от пробела 0x20 до конца)
    if (c < 0x20) return;
    
    // Вычисляем индекс символа в нашем массиве шрифта
    uint16_t char_index = c - 0x20;
    
    // Указатель на начало данных нашей буквы (26 колонок)
    const uint16_t *char_data = &CustomFont16x26[char_index * 26];

    // Цикл по ширине символа (26 вертикальных колонок)
    for (uint8_t col = 0; col < 26; col++) {
        uint16_t column_pixels = char_data[col];

        // Цикл по высоте символа (16 пикселей в каждой колонке)
        for (uint8_t row = 0; row < 16; row++) {
            // Проверяем каждый бит снизу вверх (или сверху вниз, подстроим)
            if (column_pixels & (1 << row)) {
                // Если бит равен 1 — зажигаем пиксель максимальной яркостью
                SSD1322_DrawPixel(x + col, y + row, 0x0F); 
            } else {
                // Если бит равен 0 — гасим пиксель (черный фон)
                SSD1322_DrawPixel(x + col, y + row, 0x00);
            }
        }
    }
}

// ============================================================
// НАДЁЖНАЯ РАБОТА С FLASH ДЛЯ ХРАНЕНИЯ НАСТРОЕК ЦАПА
// ============================================================

#include "stm32f10x.h"

// Структура выровнена по 4 байта (размер строго кратен 32 битам)
typedef struct {
    uint8_t input_select;      // 0 – USB, 1 – S/PDIF, 2 - TOSLINK
    uint8_t digital_filter;    // Индекс активного фильтра ЦАПа
    uint8_t volume;            // Текущая громкость
    int8_t  balance;           // Баланс каналов
    uint16_t contrast;         // Яркость/Контраст дисплея
    uint16_t reserved;         // Резерв для выравнивания структуры
} presets_def;

// Наша глобальная переменная с настройками
static presets_def preset;

// Адрес последней страницы Flash для STM32F103 (64КБ версия)
// Страница занимает 1024 байта (0x400)
#define FLASH_SETTINGS_ADDR    0x0800FC00

// Считаем количество 32-битных слов с округлением вверх
#define PRESET_WORD_CNT        ((sizeof(presets_def) + 3) / 4)

/**
  * @brief  Чтение настроек из Flash памяти
  */
void FLASH_ReadSettings(void) {
    uint32_t *source_adr = (uint32_t *)FLASH_SETTINGS_ADDR;
    uint32_t *dest_adr = (uint32_t *)&preset;

    // Если Flash чистая (все единицы 0xFFFFFFFF), инициализируем базу
    if (*source_adr == 0xFFFFFFFF) {
        preset.input_select   = 0;   // USB
        preset.digital_filter = 0;   // Первый фильтр
        preset.volume         = 40;  // 40 дБ по умолчанию
        preset.balance        = 0;   // Центр
        preset.contrast       = 0x0F;// Максимальный нибл
        return;
    }

    // Читаем побайтово/пословно сохранённые данные
    for (uint16_t p = 0; p < PRESET_WORD_CNT; p++) {
        dest_adr[p] = source_adr[p];
    }
}

/**
  * @brief  Запись настроек во Flash (стирает страницу и перезаписывает)
  */
void FLASH_WriteSettings(void) {
    uint32_t pageAdr = FLASH_SETTINGS_ADDR;
    uint32_t *source_adr = (uint32_t *)&preset;

    // Ждём, пока Flash освободится от предыдущих операций
    while (FLASH->SR & FLASH_SR_BSY);

    FLASH_Unlock();
    FLASH_ErasePage(pageAdr);
    
    // Снова ждём окончания стирания страницы
    while (FLASH->SR & FLASH_SR_BSY);

    // Пословно записываем нашу структуру в память
    for (uint16_t i = 0; i < PRESET_WORD_CNT; i++) {
        FLASH_ProgramWord(pageAdr + (i * 4), source_adr[i]);
    }
    
    FLASH_Lock();
}

int16_t last_encoder_value = 0;

// Функция возвращает: 0 - не крутили, -1 - крутанули влево, 1 - крутанули вправо
int8_t Menu_Check_Encoder(void) {
    int16_t current_value = (int16_t)Encoder_read();
    int8_t result = 0;

    // Если значение счётчика изменилось
    if (current_value != last_encoder_value) {
        // Проверяем направление по твоей функции
        if (Encoder_direction() == 0) {
            result = -1; // Влево
        } else {
            result = 1;  // Вправо
        }
        last_encoder_value = current_value; // Запоминаем текущую позицию
    }
    return result;
}

// ============================================================
//   ОБНОВЛЕННАЯ НАВИГАЦИЯ ПО КНОПКЕ ДЛЯ STM32 (В ФАЙЛ MENU.C)
// ============================================================

void ProcessButtonPress(void) {
    // 1. Если экран был затенён, кнопка просто будит его на полную яркость
    if (global_dimming_mode) {
        global_dimming_mode = 0;
        dimming_timer = GetTick(); // Сбрасываем таймер затенения
        menu_timer = GetTick();    // Сбрасываем таймер автовозврата
        // Возвращаем пикселям исходный нибл (яркость) в буфере
        // SSD1322_RefreshFullBrightness(); 
        return; 
    }

    // 2. Если звук был выключен (Mute) — включаем его обратно
    if (mute_state) {
        SSD1322_CommandWrite(0xAF); // Аппаратно включаем матрицу
        mute_state = 0;
        halt_counter = 0;
    }

    // 3. Сбрасываем таймер автовозврата на главный экран (начинаем отсчет 60 сек заново)
    menu_timer = GetTick(); 

    // 4. Логика циклического переключения экранов меню
    switch (current_menu_state) {
        
        case MENU_STATE_VOLUME:
            // Из главного экрана переходим к выбору входа
            current_menu_state = MENU_STATE_INPUT;
            SSD1322_Clear(); // Чистим экран под новые большие буквы входа
            break;

        case MENU_STATE_INPUT:
            // Из выбора входа переходим к выбору фильтра
            current_menu_state = MENU_STATE_FILTER;
            SSD1322_Clear();
            break;

        case MENU_STATE_FILTER:
            // Из фильтров переходим к настройке баланса
            current_menu_state = MENU_STATE_BALANCE;
            SSD1322_Clear();
            break;

        case MENU_STATE_BALANCE:
            // Из баланса возвращаемся обратно на главный экран громкости
            current_menu_state = MENU_STATE_VOLUME;
            SSD1322_Clear();
            break;
            
        default:
            current_menu_state = MENU_STATE_VOLUME;
            break;
    }
}
